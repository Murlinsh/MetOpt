```python
from math import sin, cos

def f(p):
    x, y = p
    return sin(x) + cos(y) + 0.1 * (x**2 + y**2)

def hooke_jeeves(f, x0, h0, alpha=0.5, eps=1e-6, max_iter=500, explain=False):
    x = x0[:]
    h = h0[:]
    n = len(x)
    k = 0

    if explain:
        print(f"Начальная точка: x = {x}, f = {f(x):.6f}")
        print(f"Начальный шаг: h = {h}, alpha = {alpha}, eps = {eps}")
        print("-" * 60)

    while max(abs(hi) for hi in h) > eps and k < max_iter:
        k += 1
        x_prev = x[:]

        if explain:
            print(f"\n=== Итерация {k} ===")
            print(f"  Текущая точка: {x}, f = {f(x):.6f}, h = {h}")

        # Исследующий поиск
        for i in range(n):
            y = x[:]
            y[i] = x[i] + h[i]
            if f(y) < f(x):
                x = y
                if explain:
                    print(f"  Шаг + по x_{i}: {y}, f = {f(y):.6f} — улучшение")
            else:
                y[i] = x[i] - h[i]
                if f(y) < f(x):
                    x = y
                    if explain:
                        print(f"  Шаг - по x_{i}: {y}, f = {f(y):.6f} — улучшение")
                elif explain:
                    print(f"  Шаги по x_{i} в +/-: улучшения нет")

        if f(x) < f(x_prev):
            if explain:
                print(f"  Исследующий поиск успешен: x = {x}, f = {f(x):.6f}")
                print(f"  Поиск по образцу:")
            while True:
                x_p = [2 * x[i] - x_prev[i] for i in range(n)]
                x_new = x_p[:]
                for i in range(n):
                    y = x_new[:]
                    y[i] = x_new[i] + h[i]
                    if f(y) < f(x_new):
                        x_new = y
                    else:
                        y[i] = x_new[i] - h[i]
                        if f(y) < f(x_new):
                            x_new = y
                if f(x_new) < f(x):
                    x_prev = x[:]
                    x = x_new[:]
                    if explain:
                        print(f"    Успешный шаг по образцу: x = {x}, f = {f(x):.6f}")
                else:
                    if explain:
                        print(f"    Шаг по образцу без улучшения — стоп")
                    break
        else:
            h = [hi * alpha for hi in h]
            if explain:
                print(f"  Уменьшаем шаг: h = {h}")

    return x, k

def run():
    print("=== Метод Хука-Дживса (Кука-Дживса) ===")
    print("1) Готовые ответы\n2) Режим объяснения")
    explain = input("Режим (1/2): ").strip() == "2"

    x0 = float(input("x0: "))
    y0 = float(input("y0: "))
    h0 = float(input("Начальный шаг h: "))
    eps = float(input("Epsilon: "))

    x_min, iters = hooke_jeeves(f, [x0, y0], [h0, h0], eps=eps, explain=explain)

    print("\n" + "=" * 40)
    print(f"Приблизительный минимум: x = {x_min[0]:.6f}, y = {x_min[1]:.6f}")
    print(f"Значение функции: f = {f(x_min):.6f}")
    print(f"Количество итераций: {iters}")

if __name__ == "__main__":
```

```python
from math import sin, cos, sqrt

def f(p):
    x, y = p
    return sin(x) + cos(y) + 0.1 * (x**2 + y**2)

def nelder_mead(f, x0, step=1.0, alpha=1.0, beta=0.5, gamma=2.0,
                eps=1e-4, max_iter=1000, explain=False):
    n = len(x0)
    simplex = [x0[:]]
    for i in range(n):
        p = x0[:]
        p[i] += step
        simplex.append(p)

    k = 0
    if explain:
        print(f"Начальный симплекс: {simplex}")
        print(f"alpha={alpha} (отраж.), beta={beta} (сжат.), gamma={gamma} (раст.), eps={eps}")
        print("-" * 60)

    while k < max_iter:
        k += 1
        simplex.sort(key=f)
        p_l, p_g, p_h = simplex[0], simplex[1], simplex[2]
        f_l, f_g, f_h = f(p_l), f(p_g), f(p_h)
        x_c = [(p_l[i] + p_g[i]) / 2 for i in range(n)]
        f_c = f(x_c)
        sigma = sqrt(((f_l - f_c) ** 2 + (f_g - f_c) ** 2 + (f_h - f_c) ** 2) / 3)

        if explain:
            print(f"\n=== Итерация {k} ===")
            print(f"  Лучшая:   {p_l}, f = {f_l:.6f}")
            print(f"  Средняя:  {p_g}, f = {f_g:.6f}")
            print(f"  Худшая:   {p_h}, f = {f_h:.6f}")
            print(f"  Центроид: {x_c}, f = {f_c:.6f}")
            print(f"  sigma = {sigma:.6f}")

        if sigma < eps:
            if explain:
                print("  Критерий сходимости выполнен — стоп")
            break

        # Отражение
        p_r = [x_c[i] + alpha * (x_c[i] - p_h[i]) for i in range(n)]
        f_r = f(p_r)
        if explain:
            print(f"  Отражение: {p_r}, f = {f_r:.6f}")

        if f_l <= f_r < f_g:
            simplex[2] = p_r
            if explain:
                print("  Отражение принято")
            continue

        if f_r < f_l:
            # Растяжение
            p_e = [x_c[i] + gamma * (p_r[i] - x_c[i]) for i in range(n)]
            f_e = f(p_e)
            if explain:
                print(f"  Растяжение: {p_e}, f = {f_e:.6f}")
            simplex[2] = p_e if f_e < f_r else p_r
            if explain:
                print(f"  Принято: {'P_e' if f_e < f_r else 'P_r'}")
            continue

        # Сжатие
        if f_r < f_h:
            p_s = [x_c[i] + beta * (p_r[i] - x_c[i]) for i in range(n)]
            if explain: print(f"  Внешнее сжатие: {p_s}")
        else:
            p_s = [x_c[i] + beta * (p_h[i] - x_c[i]) for i in range(n)]
            if explain: print(f"  Внутреннее сжатие: {p_s}")
        f_s = f(p_s)
        if explain: print(f"  f(P_s) = {f_s:.6f}")

        if f_s < min(f_r, f_h):
            simplex[2] = p_s
            if explain: print("  Сжатие принято")
        else:
            if explain: print("  Редукция симплекса к P_l")
            for i in range(1, 3):
                simplex[i] = [p_l[j] + 0.5 * (simplex[i][j] - p_l[j]) for j in range(n)]
            if explain: print(f"  Новый симплекс: {simplex}")

    simplex.sort(key=f)
    return simplex[0], k

def run():
    print("=== Метод Нелдера-Мида ===")
    print("1) Готовые ответы\n2) Режим объяснения")
    explain = input("Режим (1/2): ").strip() == "2"

    x0 = float(input("x0: "))
    y0 = float(input("y0: "))
    step = float(input("Начальный размер симплекса step: ") or 1.0)
    alpha = float(input("alpha (1.0): ") or 1.0)
    beta  = float(input("beta  (0.5): ") or 0.5)
    gamma = float(input("gamma (2.0): ") or 2.0)
    eps   = float(input("Epsilon: "))

    x_min, iters = nelder_mead(f, [x0, y0], step, alpha, beta, gamma, eps, explain=explain)

    print("\n" + "=" * 40)
    print(f"Приблизительный минимум: x = {x_min[0]:.6f}, y = {x_min[1]:.6f}")
    print(f"Значение функции: f = {f(x_min):.6f}")
    print(f"Количество итераций: {iters}")

if __name__ == "__main__":
    run()
```

## Файл 3 — `ravine.py` (метод оврагов Гельфанда-Цетлина)

```python
from math import sin, cos, sqrt

def f(p):
    x, y = p
    return sin(x) + cos(y) + 0.1 * (x**2 + y**2)

def num_grad(f, x, h=1e-6):
    return [(f([x[0] + h, x[1]]) - f([x[0] - h, x[1]])) / (2 * h),
            (f([x[0], x[1] + h]) - f([x[0], x[1] - h])) / (2 * h)]

def descend(f, x, eps=1e-5, max_iter=500):
    """Спуск к дну оврага: градиентный спуск с backtracking."""
    x = x[:]
    for _ in range(max_iter):
        g = num_grad(f, x)
        gn = sqrt(g[0] ** 2 + g[1] ** 2)
        if gn < eps:
            break
        alpha, c, rho = 0.1, 1e-4, 0.5
        while f([x[0] - alpha * g[0], x[1] - alpha * g[1]]) > f(x) - c * alpha * gn * gn:
            alpha *= rho
            if alpha < 1e-12:
                break
        x = [x[0] - alpha * g[0], x[1] - alpha * g[1]]
    return x

def ravine_method(f, x0, step=1.0, eps=1e-4, inner_eps=1e-4,
                  max_iter=100, explain=False):
    x = descend(f, x0, eps=inner_eps)

    if explain:
        print(f"Начальная точка: {x0}, f = {f(x0):.6f}")
        print(f"После спуска к дну оврага: x = {x}, f = {f(x):.6f}")
        print(f"Шаг h = {step}, eps = {eps}")
        print("-" * 60)

    h = step
    k = 0
    dir_idx = 0  # 0 → ось x, 1 → ось y

    while h > eps and k < max_iter:
        k += 1
        d = [0.0, 0.0]
        d[dir_idx] = 1.0

        if explain:
            print(f"\n=== Итерация {k} ===")
            print(f"  Дно оврага: {x}, f = {f(x):.6f}")
            print(f"  Шаг h = {h}, направление d = {d}")

        x_try = descend(f, [x[0] + h * d[0], x[1] + h * d[1]], eps=inner_eps)
        if explain:
            print(f"  Шаг +h·d и спуск: {x_try}, f = {f(x_try):.6f}")

        if f(x_try) < f(x):
            x = x_try
            if explain:
                print("  Улучшение — продолжаем в том же направлении")
        else:
            x_try = descend(f, [x[0] - h * d[0], x[1] - h * d[1]], eps=inner_eps)
            if explain:
                print(f"  Шаг −h·d и спуск: {x_try}, f = {f(x_try):.6f}")
            if f(x_try) < f(x):
                x = x_try
                if explain:
                    print("  Улучшение в обратную сторону")
            else:
                h = h / 2
                dir_idx = (dir_idx + 1) % 2
                if explain:
                    print(f"  Улучшения нет — h = {h}, меняем направление на ось {dir_idx}")

    return x, k

def run():
    print("=== Метод оврагов ===")
    print("1) Готовые ответы\n2) Режим объяснения")
    explain = input("Режим (1/2): ").strip() == "2"

    x0 = float(input("x0: "))
    y0 = float(input("y0: "))
    step = float(input("Начальный шаг h: "))
    eps  = float(input("Epsilon (внешнее): "))
    inner_eps = float(input("Epsilon (внутреннее, спуск): ") or 1e-4)

    x_min, iters = ravine_method(f, [x0, y0], step, eps, inner_eps, explain=explain)

    print("\n" + "=" * 40)
    print(f"Приблизительный минимум: x = {x_min[0]:.6f}, y = {x_min[1]:.6f}")
    print(f"Значение функции: f = {f(x_min):.6f}")
    print(f"Количество итераций: {iters}")

if __name__ == "__main__":
    run()
```

## Файл 4 — `gradient_descent.py` (3 подметода)

```python
from math import sin, cos, sqrt
import random

def f(p):
    x, y = p
    return sin(x) + cos(y) + 0.1 * (x**2 + y**2)

def grad_f(p):
    x, y = p
    return [cos(x) + 0.2 * x, -sin(y) + 0.2 * y]

def norm(v): return sqrt(sum(vi * vi for vi in v))
def dot(a, b): return sum(ai * bi for ai, bi in zip(a, b))
def sub(a, b): return [ai - bi for ai, bi in zip(a, b)]
def add(a, b): return [ai + bi for ai, bi in zip(a, b)]
def mul(v, s): return [vi * s for vi in v]

def line_search(f, x, d, g, alpha=1.0, c=1e-4, rho=0.5):
    while f(add(x, mul(d, alpha))) > f(x) + c * alpha * dot(g, d):
        alpha *= rho
        if alpha < 1e-14:
            break
    return alpha

# ---------- Наискорейший спуск ----------
def steepest_descent(f, grad_f, x0, eps=1e-6, max_iter=5000, explain=False):
    x = x0[:]
    k = 0
    while k < max_iter:
        k += 1
        g = grad_f(x)
        gn = norm(g)
        if explain:
            print(f"\n=== Итерация {k} ===")
            print(f"  x = {x}, f = {f(x):.6f}")
            print(f"  grad = {[f'{v:.6f}' for v in g]}, |grad| = {gn:.6f}")
        if gn < eps:
            if explain: print("  |grad| < eps — стоп")
            break
        d = mul(g, -1)
        alpha = line_search(f, x, d, g)
        x = add(x, mul(d, alpha))
        if explain:
            print(f"  alpha = {alpha:.6f}, новое x = {x}, f = {f(x):.6f}")
    return x, k

# ---------- Сопряжённые градиенты (Флетчер-Ривс) ----------
def conjugate_gradient(f, grad_f, x0, eps=1e-6, max_iter=5000, explain=False):
    x = x0[:]
    g = grad_f(x)
    d = mul(g, -1)
    k = 0
    while k < max_iter:
        k += 1
        gn = norm(g)
        if explain:
            print(f"\n=== Итерация {k} ===")
            print(f"  x = {x}, f = {f(x):.6f}")
            print(f"  g = {[f'{v:.6f}' for v in g]}, |g| = {gn:.6f}")
            print(f"  d = {[f'{v:.6f}' for v in d]}")
        if gn < eps:
            if explain: print("  |g| < eps — стоп")
            break
        alpha = line_search(f, x, d, g)
        x_new = add(x, mul(d, alpha))
        g_new = grad_f(x_new)
        gg_old = dot(g, g)
        beta = dot(g_new, g_new) / gg_old if gg_old > 1e-16 else 0.0
        d = add(mul(g_new, -1), mul(d, beta))
        if explain:
            print(f"  alpha = {alpha:.6f}, beta = {beta:.6f}")
            print(f"  новое x = {x_new}, f = {f(x_new):.6f}")
        x, g = x_new, g_new
    return x, k

# ---------- Стохастический градиентный спуск ----------
def stochastic_gradient_descent(f, grad_f, x0, eps=1e-6,
                                lr0=0.05, sigma=0.1, max_iter=5000, explain=False):
    """
    Примечание: SGD обычно применяют к f = (1/N)·Σ f_i (по выборке).
    Для детерминированной функции имитируем стохастичность,
    добавляя гауссов шум к градиенту на каждой итерации.
    """
    x = x0[:]
    k = 0
    while k < max_iter:
        k += 1
        g = grad_f(x)
        gn = norm(g)
        noise = [random.gauss(0, sigma) for _ in g]
        g_noisy = [gi + ni for gi, ni in zip(g, noise)]
        lr = lr0 / (1 + 0.01 * k)
        if explain:
            print(f"\n=== Итерация {k} ===")
            print(f"  x = {x}, f = {f(x):.6f}")
            print(f"  g = {[f'{v:.6f}' for v in g]}, |g| = {gn:.6f}")
            print(f"  шум = {[f'{v:.6f}' for v in noise]}")
            print(f"  g_noisy = {[f'{v:.6f}' for v in g_noisy]}, lr = {lr:.6f}")
        if gn < eps:
            if explain: print("  |g| < eps — стоп")
            break
        x = add(x, mul(g_noisy, -lr))
        if explain:
            print(f"  новое x = {x}, f = {f(x):.6f}")
    return x, k

def run():
    print("=== Градиентные методы ===")
    print("1) Наискорейший градиентный спуск")
    print("2) Метод сопряжённых градиентов")
    print("3) Стохастический градиентный спуск")
    m = input("Метод (1/2/3): ").strip()

    print("\nРежим: 1) Готовые ответы  2) Объяснение")
    explain = input("Режим (1/2): ").strip() == "2"

    x0 = float(input("x0: "))
    y0 = float(input("y0: "))
    eps = float(input("Epsilon: "))

    if m == "1":
        x_min, iters = steepest_descent(f, grad_f, [x0, y0], eps=eps, explain=explain)
    elif m == "2":
        x_min, iters = conjugate_gradient(f, grad_f, [x0, y0], eps=eps, explain=explain)
    else:
        sigma = float(input("sigma шума (0.1): ") or 0.1)
        lr0 = float(input("Начальный learning rate (0.05): ") or 0.05)
        x_min, iters = stochastic_gradient_descent(f, grad_f, [x0, y0],
                                                   eps=eps, lr0=lr0, sigma=sigma,
                                                   explain=explain)

    print("\n" + "=" * 40)
    print(f"Приблизительный минимум: x = {x_min[0]:.6f}, y = {x_min[1]:.6f}")
    print(f"Значение функции: f = {f(x_min):.6f}")
    print(f"Количество итераций: {iters}")

if __name__ == "__main__":
    run()
```

```python
from math import sin, cos, sqrt

def f(p):
    x, y = p
    return sin(x) + cos(y) + 0.1 * (x**2 + y**2)

def grad_f(p):
    x, y = p
    return [cos(x) + 0.2 * x, -sin(y) + 0.2 * y]

def hess_f(p):
    x, y = p
    return [[-sin(x) + 0.2, 0.0],
            [0.0,           -cos(y) + 0.2]]

def norm(v): return sqrt(sum(vi * vi for vi in v))
def dot(a, b): return sum(ai * bi for ai, bi in zip(a, b))
def sub(a, b): return [ai - bi for ai, bi in zip(a, b)]
def add(a, b): return [ai + bi for ai, bi in zip(a, b)]
def mul(v, s): return [vi * s for vi in v]

def solve_2x2(H, b):
    det = H[0][0] * H[1][1] - H[0][1] * H[1][0]
    if abs(det) < 1e-14:
        return [0.0, 0.0]
    return [(b[0] * H[1][1] - b[1] * H[0][1]) / det,
            (H[0][0] * b[1] - H[1][0] * b[0]) / det]

# ---------- Классический Ньютон ----------
def newton_method(f, grad_f, hess_f, x0, eps=1e-6, max_iter=100, explain=False):
    x = x0[:]
    k = 0
    while k < max_iter:
        k += 1
        g = grad_f(x)
        gn = norm(g)
        if explain:
            print(f"\n=== Итерация {k} ===")
            print(f"  x = {x}, f = {f(x):.6f}")
            print(f"  g = {[f'{v:.6f}' for v in g]}, |g| = {gn:.6f}")
        if gn < eps:
            if explain: print("  |g| < eps — стоп")
            break
        H = hess_f(x)
        if explain:
            print(f"  H = [[{H[0][0]:.6f}, {H[0][1]:.6f}], [{H[1][0]:.6f}, {H[1][1]:.6f}]]")
        d = solve_2x2(H, mul(g, -1))
        x = add(x, d)
        if explain:
            print(f"  d = {[f'{v:.6f}' for v in d]}")
            print(f"  новое x = {x}, f = {f(x):.6f}")
    return x, k

# ---------- Квазиньютоновский BFGS ----------
def bfgs(f, grad_f, x0, eps=1e-6, max_iter=500, explain=False):
    n = 2
    x = x0[:]
    H = [[1.0 if i == j else 0.0 for j in range(n)] for i in range(n)]
    g = grad_f(x)
    k = 0
    while k < max_iter:
        k += 1
        gn = norm(g)
        if explain:
            print(f"\n=== Итерация {k} ===")
            print(f"  x = {x}, f = {f(x):.6f}, |g| = {gn:.6f}")
        if gn < eps:
            if explain: print("  |g| < eps — стоп")
            break
        # d = -H·g (H аппроксимирует обратный гессиан)
        d = [-sum(H[i][j] * g[j] for j in range(n)) for i in range(n)]
        # backtracking
        alpha, c, rho = 1.0, 1e-4, 0.5
        while f(add(x, mul(d, alpha))) > f(x) + c * alpha * dot(g, d):
            alpha *= rho
            if alpha < 1e-14:
                break
        x_new = add(x, mul(d, alpha))
        g_new = grad_f(x_new)
        s = sub(x_new, x)
        y = sub(g_new, g)

        if explain:
            print(f"  d = {[f'{v:.6f}' for v in d]}, alpha = {alpha:.6f}")
            print(f"  x_new = {x_new}, f = {f(x_new):.6f}")
            print(f"  s = {[f'{v:.6f}' for v in s]}, y = {[f'{v:.6f}' for v in y]}")

        ys = dot(y, s)
        if ys > 1e-12:
            rho_k = 1.0 / ys
            V = [[(1.0 if i == j else 0.0) - rho_k * s[i] * y[j] for j in range(n)]
                 for i in range(n)]
            VH = [[sum(V[i][t] * H[t][j] for t in range(n)) for j in range(n)]
                  for i in range(n)]
            VHVT = [[sum(VH[i][t] * V[j][t] for t in range(n)) for j in range(n)]
                    for i in range(n)]
            H = [[VHVT[i][j] + rho_k * s[i] * s[j] for j in range(n)] for i in range(n)]
            if explain:
                print(f"  BFGS-обновление H выполнено")
        else:
            if explain:
                print(f"  y·s ≤ 0, обновление H пропущено")

        x, g = x_new, g_new
    return x, k
```

## 5. Что такое матрица Гессе (гессиан) — доходчиво

**Идея.** Обычная производная говорит, **как быстро меняется функция**. Вторая производная — **как быстро меняется сама производная**, то есть насколько функция «кривая» в данной точке. Для функции одной переменной `f(x)` есть просто `f''(x)`. Для функции двух переменных `f(x, y)` «вторая производная» — это уже **четыре числа**, потому что дифференцировать можно дважды по `x`, дважды по `y`, и «смешанно»: сначала по `x`, потом по `y` (и наоборот). Эти четыре числа собираются в матрицу 2×2 — её и называют **матрицей Гессе** (Hessian):

```
        ⎡ ∂²f    ∂²f ⎤
        ⎢ ───    ─── ⎥
        ⎢ ∂x²    ∂x∂y⎥
H(x,y)= ⎢            ⎥
        ⎢ ∂²f    ∂²f ⎥
        ⎢ ───    ─── ⎥
        ⎣ ∂y∂x   ∂y² ⎦
```

Обозначения читаются так: `∂²f/∂x²` — «вторая производная по x»; `∂²f/∂x∂y` — «сначала по x, потом по y» (для гладких функций это то же самое, что сначала по y, потом по x — поэтому матрица симметрична, `H[0][1] = H[1][0]`).

**Зачем она нужна в Ньютоне.** В одномерном Ньютоне мы искали ноль производной по формуле `x_new = x − f'(x)/f''(x)`. В двумерном — то же самое, только вместо деления на число делим на матрицу: `x_new = x − H⁻¹ · ∇f(x)`, где `H⁻¹` — **обратная матрица Гессе** (та самая, которую вы строили на практике), а `∇f(x)` — вектор-градиент `[∂f/∂x, ∂f/∂y]`. Смысл тот же: «идём туда, где производная обращается в ноль, с поправкой на кривизну».

**Пример — твоя функция** `f(x, y) = sin(x) + cos(y) + 0.1·(x² + y²)`:

1. **Первые производные (градиент):**
   - `∂f/∂x = cos(x) + 0.2·x` (производная `sin(x)` — это `cos(x)`; производная `0.1·x²` — это `0.2·x`; `cos(y)` от `x` не зависит, даёт 0)
   - `∂f/∂y = −sin(y) + 0.2·y` (производная `cos(y)` — это `−sin(y)`; `sin(x)` от `y` не зависит)
   - Градиент: `∇f = [cos(x) + 0.2·x,  −sin(y) + 0.2·y]`

2. **Вторые производные:**
   - `∂²f/∂x²` — дифференцируем `cos(x) + 0.2·x` по `x`: получаем `−sin(x) + 0.2`
   - `∂²f/∂x∂y` — дифференцируем `cos(x) + 0.2·x` по `y`: получаем `0` (от y ничего не зависит)
   - `∂²f/∂y∂x` — дифференцируем `−sin(y) + 0.2·y` по `x`: получаем `0`
   - `∂²f/∂y²` — дифференцируем `−sin(y) + 0.2·y` по `y`: получаем `−cos(y) + 0.2`

3. **Матрица Гессе:**
```
H(x, y) = ⎡ −sin(x) + 0.2      0      ⎤
          ⎣      0        −cos(y) + 0.2 ⎦
```
(в данном случае она диагональная — потому что функция «разделимая»; у общей функции `f(x,y)=x²y` были бы недиагональные элементы, например `H[0][1] = 2x`).

**Что программа делает сама.** SymPy умеет брать производные **символьно** — то есть выдаёт не число, а формулу. Программа попросит SymPy посчитать все четыре элемента `H`, соберёт их в матрицу 2×2, а потом через `H.inv()` получит обратную. Дальше подставит текущие `x`, `y` и получит числовую матрицу `H⁻¹(x, y)`, умножит её на градиент — и сделает шаг. Всё это я буду печатать в режиме объяснения: «градиент такой-то, матрица Гессе такая-то, обратная такая-то, шаг такой-то, новая точка такая-то».

**В шпаргалку для пользователя** напишу кратко то же самое с примером выше, чтобы если ты открыл `SYNTAX.md` — сразу увидел, откуда берутся `H[0][0]`, `H[0][1]` и т.п. (При ручном вводе градиента — а он бывает только в fallback-случае — матрицу Гессе всё равно попросим посчитать программу: если grad введён, его можно продифференцировать символьно ещё раз; если и это не выйдет, перейдём на численный гессиан.)

**Хук-Дживс — `max(|h_x|, |h_y|) < Epsilon`.**

**Нелдер-Мид — `sigma(f_l, f_g, f_h) < Epsilon`, где `sigma` — среднеквадратичное отклонение значений функции в трёх вершинах симплекса от среднего.** Было в твоём `Nelder_Mid.py`.
Логика: если значения f во всех трёх вершинах почти одинаковые — симплекс «сплющился», все точки дают примерно один и тот же результат, значит минимум найден. **Альтернатива** — останавливаться, когда **размер симплекса** стал меньше Epsilon (то есть все три точки сблизились в пространстве). Второй вариант геометрически понятнее, первый — «по значениям функции». **Мне больше нравится по размеру симплекса** (проще объяснить: «вершины почти слиплись → стоп»), но если вы на паре делали `sigma` — оставим `sigma`. Что выбираешь?

**Метод оврагов — `h < Epsilon` (внешний шаг).**
Логика: метод двигается «поперёк оврага» шагом `h`. Шаг маленький → мы уже почти на дне оврага. **Однозначно ок.**

**Наискорейший / сопряжённые градиенты / Ньютон / BFGS — `|∇f| < Epsilon`.**
Логика: в точке минимума (для гладкой f) градиент равен нулю. Когда `|∇f|` маленький — мы у цели. Это **самый стандартный критерий** для всех градиентных методов. Есть альтернатива — «шаг стал маленьким», но она подводит, когда градиент почти нулевой ещё далеко от минимума. **Рекомендую только `|∇f| < Epsilon`.**

**SGD — `|∇f| < Epsilon` по «чистому» градиенту (без шума).**
Шаг делаем по зашумлённому градиенту, но проверку сходимости — по настоящему. Иначе шум не даст никогда остановиться. **Логично, других вариантов особо нет.**

## 6(ж) SGD — что это и почему тут неоднозначно

**Что такое «стохастический».** Обычный градиентный спуск на каждой итерации считает градиент по **всей** функции и делает шаг. Стохастический (SGD) появился в машинном обучении, где функция потерь имеет вид **суммы по большой выборке**:

```
f(w) = (1/N) · Σ f_i(w)
```

Считать полный градиент `∇f = (1/N)·Σ ∇f_i` на каждой итерации дорого (N может быть миллион). Идея SGD: **на каждой итерации брать один случайный элемент (или мини-батч) `i` и делать шаг по его градиенту `∇f_i`**. Это дёшево. Градиент получается «шумный» (случайный), но в среднем он смотрит туда же, куда настоящий, и метод всё равно сходится — иногда даже быстрее выбирается из локальных ям. Отсюда «стохастический» = «случайный».

**Проблема в нашей задаче.** У нас **детерминированная функция** одной-двух переменных, никакой выборки нет. Брать «один элемент» не из чего. Поэтому «настоящий» SGD в чистом виде применить нельзя. Остаются два разумных подхода:

**(1) Имитация шумом (то, что я предлагал).**
На каждой итерации считаем обычный градиент `g = ∇f(x)`, добавляем к нему случайный гауссов шум: `g_noisy = g + N(0, σ²)` (в каждой компоненте независимо), делаем шаг `x -= lr_k · g_noisy`. Скорость обучения уменьшаем: `lr_k = lr0 / (1 + c·k)`, чтобы к концу шум «переставал мешать».
- Плюсы: работает для любой f, легко объяснить («шумный градиент, как в SGD»).
- Минусы: это **не тот** SGD, что в учебниках по ML, а «шумный градиентный спуск». Препод может спросить «а где тут мини-батч?» — придётся объяснять.
- Параметры: `σ` (амплитуда шума, по умолчанию `0.1`), `lr0` (стартовый learning rate, `0.05`), `c` (скорость затухания, `0.01`).

**(2) Разбить f на сумму «слагаемых».**
Искусственно представить `f(x, y) = f₁(x, y) + f₂(x, y)` — например, `f₁ = sin(x) + 0.1·x²`, `f₂ = cos(y) + 0.1·y²`. На каждой итерации случайно выбираем одно слагаемое и шагаем по его градиенту. Это ближе к «истинному» SGD.
- Плюсы: концептуально «правильно».
- Минусы: как автоматически разбить **произвольную** f на слагаемые — непонятно. Либо просить пользователя ввести отдельно `f1` и `f2` (лишний ввод), либо разбивать по переменным (не всегда осмысленно).

**Моя рекомендация — (1)**: это стандартный «учебный» вариант, когда SGD показывают на детерминированной функции. В коде и в объяснении будет явно написано «имитация стохастичности гауссовым шумом». В шапке `SYNTAX.md` и в режиме объяснения прямо скажем, что это не SGD по выборке, а шумный GD — чтобы не было претензий.

Если у вас на паре делали по-другому — опиши, как именно, и я подстроюсь.

## 6(д) Критерии остановки — поясняю заново, подробно

Каждый итерационный метод крутится, пока «что-то не станет маленьким». Что именно — зависит от метода.

**Хук-Дживс.** Метод двигается шагами `h_x`, `h_y` (каждый по своей оси) и когда «застревает» — уменьшает шаг. Естественный момент остановки: **когда шаг стал меньше `Epsilon`**, т.е. `max(|h_x|, |h_y|) < Epsilon`. Дальше уточнять бессмысленно. Так было в твоём `Hooke_Jeeves.py`. **Оставляем.**

**Нелдер-Мид.** Метод держит «симплекс» из 3 точек на плоскости и на каждой итерации его двигает/отражает/сжимает. Когда симплекс становится очень маленьким — все 3 вершины почти слиплись, минимум локализован. Есть два способа мерить «маленькость»:
- **по размеру симплекса** — максимальное расстояние от лучшей вершины до остальных (или сумма длин рёбер) `< Epsilon`. Геометрически наглядно: «вершины сблизились». **Мой голос за это.**
- **по разбросу значений функции в вершинах** — среднеквадратичное отклонение `f` в трёх точках `< Epsilon` (это было в твоём `Nelder_Mid.py`). Тоже стандарт, но иногда срабатывает раньше времени, если функция плоская. **Жду твой выбор: размер симплекса или sigma?**

**Овраги.** Внешний цикл: пробуем шаг `h` вдоль «поперечной» оси, спускаемся на дно оврага, повторяем. Если ни в +h, ни в −h улучшения нет — уменьшаем `h` и меняем ось. **Стоп: `h < Epsilon`.** Стандартно. **Оставляем.**

**Все градиентные (наискорейший, сопряжённые, Ньютон, BFGS).** В точке минимума гладкой функции **градиент равен нулю**. Естественный критерий: **`|∇f(x)| < Epsilon`** — норма (длина) вектора градиента стала совсем маленькой. Это самый популярный критерий. Альтернатива «шаг маленький» плохая — она останавливает метод в плато, где градиент маленький, но до минимума ещё далеко. **Оставляем только `|∇f|`.**

## 6(ж) SGD — теперь с учётом твоей лекции

Твоё описание совпадает с классическим. Разберём:

**Постановка.** Есть функция `H(w) = (1/n) · Σ H_i(w)`, `i = 1..n`. Её градиент `∇H(w) = (1/n) · Σ ∇H_i(w)`. Считать **полный** градиент (сумму всех `n` слагаемых) на каждой итерации дорого, особенно в ML, где `n` — размер выборки.

**Идея SGD.** На каждой итерации `k`:
1. **Случайно выбираем один индекс `i_k ∈ {1..n}`.**
2. Считаем градиент **одного** слагаемого `∇H_{i_k}(w^(k))` — это `g^(k)` (аппроксимация полного градиента).
3. Шаг: `w^(k+1) = w^(k) − α_k · g^(k)`.

Ключевое свойство: **в среднем** `E[g^(k)] = ∇H(w^(k))` (если правильно нормировать). То есть направление «в среднем правильное», но каждый конкретный шаг — шумный. Отсюда «стохастический» = «случайный выбор одного слагаемого».

**Почему это работает, несмотря на шум.** Классическая теорема: если `α_k` убывает (`Σ α_k = ∞`, `Σ α_k² < ∞`), SGD сходится к стационарной точке почти наверное. Это причина, по которой в SGD скорость обучения **уменьшают** со временем: `α_k = α_0 / (1 + c·k)` — шум со временем «гасится».

**Как это применить к нашей задаче.** У нас `f(x, y)` — детерминированная формула, а не сумма по выборке. Но SymPy умеет **разложить любое выражение на сумму слагаемых**: `sympy.Add.make_args(f_expr)` выдаёт кортеж слагаемых. Например:
```
f = sin(x) + cos(y) + 0.1*x**2 + 0.1*y**2
   → [sin(x), cos(y), 0.1*x**2, 0.1*y**2]   (n = 4 слагаемых)
```
Дальше применяем SGD буквально:
1. Берём **случайное слагаемое** `H_i`.
2. Считаем его градиент `g = ∇H_i(x, y)` — формула символьно, потом численно в текущей точке. (Для `sin(x)` → `[cos(x), 0]`; для `cos(y)` → `[0, −sin(y)]` и т.д.)
3. Шаг `x -= α_k · g`. Но чтобы оценка была несмещённой для `∇f`, **умножаем на `n`** (`n · g`): потому что `E[∇H_{i_k}] = (1/n)·∇f`, значит `n·E[∇H_{i_k}] = ∇f`. Иначе метод в среднем ползёт в `n` раз медленнее (что, кстати, тоже иногда делают — «уменьшенный SGD»).
4. `α_k = α_0 / (1 + c·k)` — стандартное убывание.

**Про «масштабирование от 0 до 1 или от -1 до 1».** В ML это чаще всего **нормализация признаков** (данных): чтобы все входные переменные были в одном масштабе, иначе шаг по одной будет огромный, а по другой — крошечный. В нашем случае это **препроцессинг**: масштабировать начальную точку и границы (если они есть). Реализовать можно, но для тестовой `f(x, y) = sin(x) + cos(y) + 0.1·(x² + y²)` масштаб и так разумный. Предлагаю **сделать этот пункт опциональным**: в настройках SGD будет галочка «Нормализовать вход к [−1, 1]»; по умолчанию выключена; в объяснении пишем «если входные признаки разного масштаба — включите нормализацию». Ок, или обязательно включать?

**Что делать, если разложение даёт одно слагаемое** (например, `f = (x-1)**2 + (y-2)**2` — это два слагаемых `(x-1)**2` и `(y-2)**2`, но `f = exp(-x) · sin(x)` — одно слагаемое). Тогда «случайно выбирать» не из чего, SGD = обычный GD. Так и объясним: «функция не представима в виде суммы, вырождается в обычный градиентный спуск».

**Итого мой план для SGD:**
- SymPy разбивает f на слагаемые.
- Если `n > 1` — истинный SGD по одному слагаемому (с нормировкой `n` для несмещённости).
- Если `n = 1` — SGD вырождается в GD, так и пишем.
- Критерий остановки — по **полному** `|∇f(x)| < Epsilon` (по чистому, не по шумному — иначе шум не даст остановиться).
- Параметры: `α_0` (start learning rate, по умолчанию `0.05`), `c` (скорость затухания, `0.01`), опционально нормализация.

- Радиокнопки: **«Готовый ответ»** / **«Объяснение по итерациям»**.
- Кнопки: **«Далее»** (запуск расчёта и переход на страницу 4), «Назад» → страница 2.

**Страница 4 — результат.**
- Верх — `QLabel` с итогом: «Минимум: x = ..., y = ..., f = ..., итераций N».
- Основное поле — `QTextEdit` (read-only, моноширинный шрифт) с логом итераций (в режиме объяснения) или пустое (в режиме ответа).
- `QLabel` «Файл сохранён: `<полный путь>`».
- Кнопки:
  - **«Тот же метод, новая задача»** → страница 2, поля предзаполнены последними вводными (из `last_input.json`).
  - **«Сменить метод»** → страница 1, поля на странице 2 сохраняют последние вводные.
  - «Выход».

## 7. Поток данных

1. Старт приложения → читаем `last_input.json` (если есть) → страница 1.
2. Пользователь выбирает метод → страница 2, поля заполняются: метод-специфичные — значениями по умолчанию (если в `last_input.json` того же метода уже были — берём из него); `f, x0, y0, epsilon` — из `last_input.json`, если есть.
3. Нажатие «Далее» → парсинг f (и, если SymPy не справился — диалог fallback для градиента). После успеха: запись в `last_input.json`, переход на страницу 3.
4. Выбор режима → «Далее» → запуск `MethodSpec.run(...)` → формирование `RunResult` → запись `results/<iso>.txt` → переход на страницу 4.
5. На странице 4 — вывод лога и итога. Кнопки возврата (см. выше).

## 9. Критерии остановки (зафиксировано)

| Метод | Условие |
|---|---|
| Хук-Дживс | `max(|h_x|, |h_y|) ≤ Epsilon` |
| Нелдер-Мид | `sigma ≤ Epsilon`, где `sigma = sqrt( ((f_l−f_c)² + (f_g−f_c)² + (f_h−f_c)²) / 3 )`, `f_c = f(центроид)` |
| Овраги | `h ≤ Epsilon` (внешний шаг) |
| Наискорейший | `‖∇f‖ ≤ Epsilon` |
| Сопряжённые градиенты | `‖∇f‖ ≤ Epsilon` |
| SGD | `‖∇f‖ ≤ Epsilon` (**чистый** градиент, не шумный) |
| Классический Ньютон | `‖∇f‖ ≤ Epsilon` |
| Бройден | `‖∇f‖ ≤ Epsilon` |
| BFGS | `‖∇f‖ ≤ Epsilon` |

Все методы также останавливаются по `max_iter` (5000 по умолчанию; задаётся константой в коде, не пользователем, чтобы не плодить поля).

## 10. Обработка ввода функции

1. Пользователь вводит строку в поле «Функция f(x, y)».
2. При нажатии «Далее» на странице 2 вызывается `parse_function(s, ["x", "y"])`.
3. Если SymPy справился — вызывается `build_gradient` (символьно). Если и он справился — `build_hessian` (символьно, для Ньютона).
4. Для методов, которым нужен только f (Хук-Дживс, Нелдер-Мид, овраги) — градиент не строится, поля не спрашиваются. Если SymPy не смог и распарсить f — ошибка.
5. Для градиентных/ныотоновских методов: если `build_gradient` вернул None — показываем fallback-диалог (п.11), пользователь вводит градиент вручную; этот ввод тоже парсится и проверяется.
6. Гессиан для Ньютона: сначала символьно из f (или из ручного градиента, продифференцировав его ещё раз); если не вышло — численно.

**Сообщение об ошибке парсинга** содержит:
- какое место не распознано;
- напоминание про `**` (не `^`) и явное умножение;
- отсылку к кнопке «Синтаксис».

## 11. Fallback для градиента

Если `build_gradient` вернул None, показывается `QDialog` с таким текстом (заголовок окна — «Ручной ввод градиента»):

> **Не удалось построить градиент автоматически.**
>
> Введите градиент в формате: `[∂f/∂x, ∂f/∂y]`
>
> **Что такое градиент.** Для функции двух переменных `f(x, y)` — это вектор из двух частных производных:
> - **∂f/∂x** — производная по `x`, при этом `y` считаем константой;
> - **∂f/∂y** — производная по `y`, при этом `x` считаем константой.
>
> **Пример.** Для `f(x, y) = sin(x) + cos(y) + 0.1·(x² + y²)`:
> - `∂f/∂x = cos(x) + 0.2·x` — дифференцируем по x: `sin(x)` → `cos(x)`, `0.1·x²` → `0.2·x`, `cos(y)` даёт 0;
> - `∂f/∂y = −sin(y) + 0.2·y` — дифференцируем по y: `cos(y)` → `−sin(y)`, `0.1·y²` → `0.2·y`, `sin(x)` даёт 0;
> - В поле ввода пишем: `[cos(x) + 0.2*x, -sin(y) + 0.2*y]`
>
> **Синтаксис и пробелы.**
> - Степень — `**` (не `^`). Умножение — явное (`2*x`). Функции: `sin, cos, exp, log, sqrt`. Константы: `pi`, `E`.
> - Пробелы внутри выражения допускаются в любом количестве.
> - Квадратные скобки вокруг всего списка — **обязательны**, внутри — два выражения через запятую.
> - Полное руководство — кнопка «Синтаксис» в верхней панели.

Под полем — кнопки «OK» / «Отмена». При «Отмена» — возврат на страницу 2 (расчёт не запускается).

## 12. Сохранение последних вводных (`last_input.json`)

Файл в корне проекта, в `.gitignore`, обновляется **при каждом успешном нажатии «Далее» со страницы 2** — вне зависимости от того, какую кнопку на странице 4 пользователь выберет позже. Это позволяет после «Сменить метод» (возврат на страницу 1) увидеть на странице 2 те же вводные при выборе любого другого метода, а после «Тот же метод, новая задача» — увидеть те же вводные для текущего метода.

**Схема:**
```json
{
  "last_method_id": "bfgs",
  "common": {
    "f_expr": "sin(x) + cos(y) + 0.1*(x**2 + y**2)",
    "x0": 1.0,
    "y0": 1.0,
    "epsilon": 0.0001,
    "gradient_manual": null,
    "mode": "explain"
  },
  "per_method": {
    "hooke_jeeves": {"h": 0.5, "alpha": 0.5},
    "nelder_mead": {"step": 1.0, "alpha": 1.0, "beta": 0.5, "gamma": 2.0},
    "ravine":      {"h": 0.5, "inner_eps": 0.0001},
    "sgd":         {"alpha0": 0.05, "c": 0.01, "normalize": false}
  }
}
```

При старте приложения:
- Если файла нет или он битый — игнорируем, страница 2 пустая (только дефолты специфических полей).
- Если есть — на странице 2 поля `common` заполнены из `last_input.json`; при выборе метода подтягиваются `per_method[method_id]` как значения по умолчанию для специфических полей (поверх общих дефолтов).

## 13. Формат файла результатов

Путь: `results/<ISO-datetime>.txt`, где `<ISO-datetime>` — `YYYY-MM-DDTHH-MM-SS` (двоеточия заменены на `-` для совместимости с Windows).

Пример полного имени: `results/2026-11-08T14-35-12.txt`.

**Шаблон файла (успешный расчёт):**
```
=== МЕТАДАННЫЕ ===
Дата и время: 2026-11-08T14-35-12
Метод: Ньютон → Классический
Режим: Объяснение по итерациям
Функция: sin(x) + cos(y) + 0.1*(x**2 + y**2)
Градиент: [cos(x) + 0.2*x, -sin(y) + 0.2*y]   (символьный)
Гессиан: [[-sin(x) + 0.2, 0], [0, -cos(y) + 0.2]]  (символьный)
x0 = 1.0, y0 = 1.0
Epsilon = 0.0001
Спец. параметры: <нет или перечень>

=== ИТЕРАЦИИ ===
<лог итераций, если режим «объяснение»; если «ответ» — раздел пуст>

=== РЕЗУЛЬТАТ ===
x* = (-1.570796, 0.000123)
f(x*) = -1.099876
Итераций: 7
```

**Шаблон файла (ошибка):**
```
=== ОШИБКА ===
Дата и время: 2026-11-08T14-35-12
Метод: <id или «не выбран»>
Функция (введённая как есть): <строка>
Поле, в котором ошибка: <имя поля>
Сообщение: <человеко-читаемое описание>
```

## 14. Обработка ошибок

| Где | Что происходит |
|---|---|
| Парсинг f на странице 2 | `QMessageBox.critical` + запись в `results/<iso>.txt` с разделом `=== ОШИБКА ===` + возврат на страницу 2 |
| Fallback-диалог градиента: пусто/невалидно | Тот же сценарий |
| Расчёт (исключение внутри метода) | `QMessageBox.critical` с текстом исключения + сохранение в `results/<iso>.txt` как ошибка |
| Достигнут `max_iter` без сходимости | Не ошибка: сохраняем результат с пометкой «достигнут предел итераций» в метаданных; в логе — предупреждение |
| Матрица Гессе вырождена (det ≈ 0) | Пробуем «демпфирование»: `H + εI`; если не помогло — возвращаем ошибку |

## 15. `SYNTAX.md` — содержание

Файл открывается кнопкой «Синтаксис» в тулбаре через системный редактор. Содержание:

1. **Переменные:** только `x` и `y`.
2. **Степень — `**`, не `^`.** Умножение — **явное**: `2*x`, не `2x`.
3. **Список функций:** `sin, cos, tan, asin, acos, atan, exp, log (натуральный), sqrt, abs`.
4. **Константы:** `pi`, `E` (число Эйлера).
5. **Пробелы допускаются** в любом количестве.
6. **Примеры:**
   - `sin(x) + cos(y) + 0.1*(x**2 + y**2)`
   - `(x - 1)**2 + (y + 2)**2`
   - `exp(-x) * sin(x) + 0.1*x**2`
7. **Как строятся производные:** программа берёт их **автоматически** через SymPy. Если не получится — попросит ввести градиент вручную в формате `[∂f/∂x, ∂f/∂y]` с подробным напоминанием.
8. **Ручной градиент — пример:** для `sin(x) + cos(y) + 0.1*(x**2 + y**2)` → `[cos(x) + 0.2*x, -sin(y) + 0.2*y]`.
9. **Что такое гессиан** (кратко): матрица вторых производных; для Ньютона программа строит её сама, пользователь ничего не вводит.

## 17. Принятые окончательно решения (для протокола)

- PyQt6, SymPy, стандартный `json`.
- Одно окно, `QStackedWidget`, 4 страницы, тулбар с «Синтаксис» / «Папка результатов» / «О программе».
- Никакой подмены синтаксиса на лету: `^` → ошибка, `e` → ошибка (только `E`).
- Имя файла результатов — только ISO-дата-время, метод — внутри. Папка `results/` рядом со скриптом, путь показывается в статусной строке и на странице 4.
- Ошибки тоже сохраняются в `results/<iso>.txt` с разделом `=== ОШИБКА ===`.
- `last_input.json` — в корне проекта, в `.gitignore`, обновляется на каждом успешном «Далее» со страницы 2.
- Две кнопки на странице 4: «Тот же метод, новая задача» → страница 2 с предзаполнением; «Сменить метод» → страница 1; вводные на странице 2 сохраняются при любом сценарии.
- Критерии остановки — см. таблицу п.9 (Нелдер-Мид по `sigma`, все градиентные — по `‖∇f‖`).
- SGD — по схеме с лекции: разложение f на слагаемые, случайный выбор, аппроксимация градиента, `α_k` убывает; нормализация входа — опциональная галочка. При `n = 1` — вырождается в GD, так и пишем.
- Квазиньютоновские методы — **два**: Бройден (rank-1 update) и BFGS (rank-2 update).
